4pub fn require_strict(times: &[i64]) -> Result<()> {
5    let bad = if times.len() < 2 {
6        false
7    } else {
8        let mut left = times[..times.len() - 1].chunks_exact(8);
9        let mut right = times[1..].chunks_exact(8);
10        let mut bad = false;
11        for (a, b) in left.by_ref().zip(right.by_ref()) {
12            if (a[0] >= b[0])
13                | (a[1] >= b[1])
14                | (a[2] >= b[2])
15                | (a[3] >= b[3])
16                | (a[4] >= b[4])
17                | (a[5] >= b[5])
18                | (a[6] >= b[6])
19                | (a[7] >= b[7])
20            {
21                bad = true;
22                break;
23            }
24        }
25        bad || left
26            .remainder()
27            .iter()
28            .zip(right.remainder())
29            .any(|(a, b)| a >= b)
30    };
31    if bad {
32        return Err(invalid("source timestamps must be strictly increasing; inspect before resolving duplicates or reversals"));
33    }
34    Ok(())
35}

37/// Rounded rational-rate grid in [start, end), with no epoch float conversion.
38pub fn uniform_grid(
39    start: i64,
40    end: i64,
41    rate_num: u32,
42    rate_den: u32,
43    max_points: usize,
44) -> Result<Vec<i64>> {
45    if start > end || rate_num == 0 || rate_den == 0 || max_points == 0 {
46        return Err(invalid(
47            "ordered bounds, positive rational rate and max_points required",
48        ));
49    }
50    let period_num = 1_000_000_000i128 * i128::from(rate_den);
51    let rate = i128::from(rate_num);
52    if period_num < rate {
53        return Err(invalid("rate exceeds nanosecond resolution"));
54    }
55    let mut result = Vec::new();
56    let whole = period_num / rate;
57    let fraction = (period_num % rate) as u64;
58    let rate = u64::from(rate_num);
59    let mut residual = rate / 2;
60    let mut t = i128::from(start);
61    for i in 0..=max_points {
62        if t >= i128::from(end) {
63            return Ok(result);
64        }
65        if i == max_points {
66            return Err(invalid("grid exceeds max_points"));
67        }
68        result.push(i64::try_from(t).map_err(|_| Error::Overflow)?);
69        t += whole;
70        residual += fraction;
71        if residual >= rate {
72            t += 1;
73            residual -= rate;
74        }
75    }
76    unreachable!()
77}

79/// Explicit ppb correction around `origin`, then offset. Never estimates clocks.
80pub fn map_clock(times: &[i64], origin: i64, offset: i64, rate_ppb: i64) -> Result<Vec<i64>> {
81    if rate_ppb == 0 && offset == 0 {
82        return Ok(times.to_vec());
83    }
84    if rate_ppb == 0 {
85        return times
86            .iter()
87            .map(|&t| t.checked_add(offset).ok_or(Error::Overflow))
88            .collect();
89    }
90    let scale = 1_000_000_000i128 + i128::from(rate_ppb);
91    if scale <= 0 {
92        return Err(invalid("clock scale must be positive"));
93    }
94    times
95        .iter()
96        .map(|&t| {
97            let delta = (i128::from(t) - i128::from(origin)) * scale;
98            let rounded = if delta >= 0 {
99                (delta + 500_000_000) / 1_000_000_000
100            } else {
101                -((-delta + 500_000_000) / 1_000_000_000)
102            };
103            i64::try_from(i128::from(origin) + rounded + i128::from(offset))
104                .map_err(|_| Error::Overflow)
105        })
106        .collect()
107}

124/// Inspect original order; positive-interval summaries never hide reversals.
125pub fn inspect_time(times: &[i64], max_gap_ns: i64) -> Result<TimeStats> {
126    if max_gap_ns < 0 {
127        return Err(invalid("max_gap_ns must be nonnegative"));
128    }
129    if times.len() >= 2 {
130        let dt = i128::from(times[1]) - i128::from(times[0]);
131        if dt > 0
132            && times
133                .windows(2)
134                .all(|w| i128::from(w[1]) - i128::from(w[0]) == dt)
135        {
136            let dt = i64::try_from(dt).map_err(|_| Error::Overflow)?;
137            return Ok(TimeStats {
138                samples: times.len(),
139                first_ns: times.first().copied(),
140                last_ns: times.last().copied(),
141                reversals: 0,
142                duplicates: 0,
143                gaps: if dt > max_gap_ns { times.len() - 1 } else { 0 },
144                positive_interval_min_ns: Some(dt),
145                positive_interval_median_ns: Some(dt as f64),
146                positive_interval_p95_ns: Some(dt as f64),
147                positive_interval_max_ns: Some(dt),
148                estimated_hz: Some(1e9 / dt as f64),
149            });
150        }
151    }
152    let mut positive = Vec::new();
153    let (mut reversals, mut duplicates, mut gaps) = (0, 0, 0);
154    for pair in times.windows(2) {
155        let dt = i128::from(pair[1]) - i128::from(pair[0]);
156        if dt < 0 {
157            reversals += 1;
158        } else if dt == 0 {
159            duplicates += 1;
160        } else {
161            if dt > i128::from(max_gap_ns) {
162                gaps += 1;
163            }
164            positive.push(i64::try_from(dt).map_err(|_| Error::Overflow)?);
165        }
166    }
167    positive.sort_unstable();
168    let median = quantile(&positive, 0.5);
169    Ok(TimeStats {
170        samples: times.len(),
171        first_ns: times.first().copied(),
172        last_ns: times.last().copied(),
173        reversals,
174        duplicates,
175        gaps,
176        positive_interval_min_ns: positive.first().copied(),
177        positive_interval_median_ns: median,
178        positive_interval_p95_ns: quantile(&positive, 0.95),
179        positive_interval_max_ns: positive.last().copied(),
180        estimated_hz: median.map(|dt| 1e9 / dt),
181    })
182}

184fn quantile(sorted: &[i64], q: f64) -> Option<f64> {
185    if sorted.is_empty() {
186        return None;
187    }
188    let index = (sorted.len() - 1) as f64 * q;
189    let a = index.floor() as usize;
190    let b = index.ceil() as usize;
191    Some(sorted[a] as f64 + (sorted[b] as f64 - sorted[a] as f64) * (index - a as f64))
192}

202pub fn align_timestamps(
203    source: &[i64],
204    target: &[i64],
205    nearest: bool,
206    max_past_ns: i64,
207    max_future_ns: i64,
208) -> Result<TimeAlignment> {
209    require_strict(source)?;
210    if max_past_ns < 0 || max_future_ns < 0 || target.windows(2).any(|w| w[1] < w[0]) {
211        return Err(invalid(
212            "nonnegative tolerances and ordered target required",
213        ));
214    }
215    let mut result = TimeAlignment {
216        indices: vec![-1; target.len()],
217        offsets_ns: vec![0; target.len()],
218        valid: vec![false; target.len()],
219    };
220    let mut next = target
221        .first()
222        .map_or(0, |&t| source.partition_point(|&v| v <= t));
223    for (i, &t) in target.iter().enumerate() {
224        while next < source.len() && source[next] <= t {
225            next += 1;
226        }
227        let past = next
228            .checked_sub(1)
229            .filter(|&j| i128::from(t) - i128::from(source[j]) <= i128::from(max_past_ns));
230        let future = (nearest && next < source.len())
231            .then_some(next)
232            .filter(|&j| i128::from(source[j]) - i128::from(t) <= i128::from(max_future_ns));
233        let chosen = match (past, future) {
234            (Some(a), Some(b)) => Some(
235                if i128::from(t) - i128::from(source[a]) <= i128::from(source[b]) - i128::from(t) {
236                    a
237                } else {
238                    b
239                },
240            ),
241            (a, None) => a,
242            (None, b) => b,
243        };
244        if let Some(index) = chosen {
245            result.indices[i] = index as i64;
246            result.offsets_ns[i] = i64::try_from(i128::from(source[index]) - i128::from(t))
247                .map_err(|_| Error::Overflow)?;
248            result.valid[i] = true;
249        }
250    }
251    Ok(result)
252}

254/// Expand nested samples using explicit nanosecond offsets, without guessing fields.
255pub fn expand_batch_timestamps(
256    message_times: &[i64],
257    counts: &[usize],
258    offsets: &[i64],
259) -> Result<Vec<i64>> {
260    let count = counts
261        .iter()
262        .try_fold(0usize, |sum, &n| sum.checked_add(n).ok_or(Error::Overflow))?;
263    if message_times.len() != counts.len() || count != offsets.len() {
264        return Err(invalid("batch counts do not match timestamps and offsets"));
265    }
266    let mut result = offsets.to_vec();
267    let mut index = 0;
268    for (&time, &count) in message_times.iter().zip(counts) {
269        for value in &mut result[index..index + count] {
270            *value = time.checked_add(*value).ok_or(Error::Overflow)?;
271        }
272        index += count;
273    }
274    Ok(result)
